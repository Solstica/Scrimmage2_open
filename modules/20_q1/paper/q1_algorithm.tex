\begin{algorithm}[H]
\SetAlgoVlined
\SetAlgoNlRelativeSize{-1}
\SetInd{0.55em}{1.05em}
\SetKwInput{KwIn}{Input}
\SetKwInput{KwInit}{Initialize}
\SetKwInput{KwOut}{Output}
\SetKwFor{For}{for}{do}{end for}
\SetKwIF{If}{ElseIf}{Else}{if}{then}{else if}{else}{end if}
\SetKw{KwRet}{return}
\caption{问题一双光束理论反射率正演算法}
\KwIn{波数序列$\{\sigma_j\}_{j=1}^{M}$，入射角$i$，厚度$d$，材料参数$N,n_3,\boldsymbol\beta$}
\KwInit{$n_1\leftarrow1.0003$，$m^*\leftarrow0.28m_0$，建立空反射率序列$\mathcal R$}
\For{$j=1,2,\ldots,M$}{
计算波长$\lambda_j\leftarrow10^4/\sigma_j$\tcp*[r]{逐点计算}
由Sellmeier模型计算晶格介电函数$\varepsilon_{\mathrm L}(\lambda_j)$\;
由Drude弱吸收修正计算$n_{2,j}$\;
\If{$n_{2,j}^2\le0$}{
\KwRet 无效参数组\tcp*[r]{可行性检查}
}
由Snell定律计算折射角$\gamma_j$\;
分别计算$s,p$偏振的$r_{12},t_{12},r_{23},t_{21}$\;
计算传播相位$\Phi_j\leftarrow4\pi\times10^{-4}\sigma_j d n_{2,j}\cos\gamma_j$\;
\For{$u\in\{s,p\}$}{
计算$\rho_u\leftarrow r_{12}^{(u)}+t_{12}^{(u)}r_{23}^{(u)}t_{21}^{(u)}e^{-\mathrm i\Phi_j}$\;
}
计算$R_j\leftarrow\bigl(|\rho_s|^2+|\rho_p|^2\bigr)/2$并写入$\mathcal R$\tcp*[r]{非偏振平均}
}
\KwRet $\mathcal R=\{R_j\}_{j=1}^{M}$\;
\KwOut{与输入波数逐点对应的非偏振理论反射率序列$\mathcal R$}
\end{algorithm}
