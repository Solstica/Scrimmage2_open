\textbf{Algorithm 1}\quad 问题一双光束理论反射率正演算法
\vspace{0.45ex}
\hrule
\vspace{0.65ex}
\textbf{Input:} 波数序列$\{\sigma_j\}_{j=1}^{M}$，入射角$i$，厚度$d$，载流子浓度$N$，衬底折射率$n_3$，Sellmeier参数$\boldsymbol\beta$；\\[-0.2ex]
\textbf{Initialize:} $n_1\leftarrow1.0003$，$m^*\leftarrow0.28m_0$，建立空反射率序列$\mathcal R$；
\vspace{0.45ex}

\begin{tabularx}{\linewidth}{@{}r@{\hspace{0.8em}}X@{}}
1: & \textbf{for} $j=1,2,\ldots,M$ \textbf{do} \\
2: & \quad 计算波长$\lambda_j\leftarrow10^4/\sigma_j$； \\
3: & \quad 由Sellmeier模型计算晶格介电函数$\varepsilon_{\mathrm L}(\lambda_j)$； \\
4: & \quad 由Drude弱吸收修正计算$n_{2,j}$；若$n_{2,j}^2\le0$，则判定该参数组无效； \\
5: & \quad 由Snell定律计算折射角$\gamma_j$； \\
6: & \quad 分别计算$s,p$偏振的$r_{12},t_{12},r_{23},t_{21}$； \\
7: & \quad 计算传播相位$\Phi_j\leftarrow4\pi\times10^{-4}\sigma_j d n_{2,j}\cos\gamma_j$； \\
8: & \quad \textbf{for} $u\in\{s,p\}$ \textbf{do} \\
9: & \qquad 计算$\rho_u\leftarrow r_{12}^{(u)}+t_{12}^{(u)}r_{23}^{(u)}t_{21}^{(u)}e^{-\mathrm i\Phi_j}$； \\
10: & \quad \textbf{end for} \\
11: & \quad 计算$R_j\leftarrow\bigl(|\rho_s|^2+|\rho_p|^2\bigr)/2$并写入$\mathcal R$； \\
12: & \textbf{end for} \\
13: & \textbf{return} $\mathcal R=\{R_j\}_{j=1}^{M}$；
\end{tabularx}

\vspace{0.45ex}
\textbf{Output:} 与输入波数逐点对应的非偏振理论反射率序列$\mathcal R$。
